# ===================== IMPORTS =====================
import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision.transforms as transforms
from torchvision import models
from torchvision.datasets import ImageFolder

import numpy as np
from PIL import Image
import cv2
import matplotlib.pyplot as plt

from sklearn.metrics import (
    accuracy_score,
    confusion_matrix,
    classification_report,
    roc_curve,
    auc
)
from sklearn.preprocessing import label_binarize

# ===================== CONFIG =====================
SEED = 42
NUM_CLASSES = 3
BATCH_SIZE = 8
EPOCHS = 25
LR = 1e-3

torch.manual_seed(SEED)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# ===================== IMAGE ENHANCEMENT =====================
def enhance(img):
    img = np.array(img)
    lab = cv2.cvtColor(img, cv2.COLOR_RGB2LAB)
    l, a, b = cv2.split(lab)
    clahe = cv2.createCLAHE(2.0, (8,8))
    l = clahe.apply(l)
    lab = cv2.merge((l,a,b))
    return Image.fromarray(cv2.cvtColor(lab, cv2.COLOR_LAB2RGB))

# ===================== CUSTOM DATASET =====================
class EnhancedImageFolder(ImageFolder):
    def __getitem__(self, index):
        path, label = self.samples[index]
        img = Image.open(path).convert("RGB")
        img = enhance(img)
        img = self.transform(img)
        return img, label

# ===================== DATA =====================
train_dir = "/kaggle/working/ricendeficiency/ricendeficiency/RicePlantNutrientsLack/Train"
test_dir  = "/kaggle/working/ricendeficiency/ricendeficiency/RicePlantNutrientsLack/Test"

tf = transforms.Compose([
    transforms.Resize((224,224)),
    transforms.ToTensor()
])

train_ds = EnhancedImageFolder(train_dir, transform=tf)
test_ds  = EnhancedImageFolder(test_dir, transform=tf)

class_names = train_ds.classes

train_loader = torch.utils.data.DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True)
test_loader  = torch.utils.data.DataLoader(test_ds, batch_size=BATCH_SIZE)

# ===================== CNN BACKBONE =====================
class CNNBackbone(nn.Module):
    def __init__(self, num_classes):
        super().__init__()
        base = models.efficientnet_b0(weights="IMAGENET1K_V1")
        self.features = base.features
        self.pool = nn.AdaptiveAvgPool2d(1)
        self.fc = nn.Linear(1280, num_classes)

    def forward(self, x):
        fmap = self.features(x)
        pooled = self.pool(fmap).flatten(1)
        logits = self.fc(pooled)
        return fmap, logits

# ===================== SIMPLE GAT =====================
class GATLayer(nn.Module):
    def __init__(self, in_dim, out_dim):
        super().__init__()
        self.fc = nn.Linear(in_dim, out_dim, bias=False)
        self.attn = nn.Linear(out_dim * 2, 1, bias=False)

    def forward(self, x):
        h = self.fc(x)
        N = h.size(0)

        h1 = h.unsqueeze(1).repeat(1, N, 1)
        h2 = h.unsqueeze(0).repeat(N, 1, 1)
        a_input = torch.cat([h1, h2], dim=-1)

        e = self.attn(a_input).squeeze(-1)
        alpha = torch.softmax(e, dim=1)

        return torch.matmul(alpha, h).mean(0)

# ===================== CNN + GAT =====================
class CNN_GAT(nn.Module):
    def __init__(self, num_classes):
        super().__init__()
        self.cnn = CNNBackbone(num_classes)
        self.gat = GATLayer(1280, 512)
        self.gat_fc = nn.Linear(512, num_classes)

    def forward(self, x):
        fmap, cnn_logits = self.cnn(x)

        nodes = F.adaptive_avg_pool2d(fmap, (4,4))
        nodes = nodes.permute(0,2,3,1).reshape(x.size(0), -1, 1280)

        gat_logits = []
        for i in range(x.size(0)):
            gfeat = self.gat(nodes[i])
            gat_logits.append(self.gat_fc(gfeat))

        return cnn_logits, torch.stack(gat_logits)

# ===================== GEOMETRIC MEAN =====================
def geometric_mean(probs, eps=1e-9):
    probs = torch.stack(probs)
    return torch.exp(torch.mean(torch.log(probs + eps), dim=0))

# ===================== TRAIN =====================
model = CNN_GAT(NUM_CLASSES).to(device)
optimizer = torch.optim.Adam(model.parameters(), lr=LR)
criterion = nn.CrossEntropyLoss()

train_losses = []

model.train()
for epoch in range(EPOCHS):
    epoch_loss = []
    for imgs, labels in train_loader:
        imgs, labels = imgs.to(device), labels.to(device)

        cnn_logits, gat_logits = model(imgs)
        loss = criterion(cnn_logits, labels) + criterion(gat_logits, labels)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        epoch_loss.append(loss.item())

    train_losses.append(np.mean(epoch_loss))
    print(f"Epoch {epoch+1}/{EPOCHS} | Loss: {train_losses[-1]:.4f}")

# ===================== LOSS CURVE =====================
plt.figure()
plt.plot(train_losses, marker='o')
plt.title("Training Loss Curve")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.grid()
plt.show()

# ===================== EVALUATION =====================
model.eval()
y_true, y_pred, y_prob = [], [], []

with torch.no_grad():
    for imgs, labels in test_loader:
        imgs = imgs.to(device)

        cnn_logits, gat_logits = model(imgs)
        cnn_prob = torch.softmax(cnn_logits, dim=1)
        gat_prob = torch.softmax(gat_logits, dim=1)

        final_prob = geometric_mean([cnn_prob, gat_prob])
        preds = torch.argmax(final_prob, dim=1)

        y_true.extend(labels.numpy())
        y_pred.extend(preds.cpu().numpy())
        y_prob.extend(final_prob.cpu().numpy())

acc = accuracy_score(y_true, y_pred)
print(f"\nFINAL TEST ACCURACY (CNN ⊙ GAT): {acc*100:.2f}%")

# ===================== CONFUSION MATRIX =====================
cm = confusion_matrix(y_true, y_pred)

plt.figure()
plt.imshow(cm)
plt.title("Confusion Matrix")
plt.colorbar()
plt.xticks(range(NUM_CLASSES), class_names, rotation=45)
plt.yticks(range(NUM_CLASSES), class_names)
plt.xlabel("Predicted")
plt.ylabel("True")

for i in range(NUM_CLASSES):
    for j in range(NUM_CLASSES):
        plt.text(j, i, cm[i, j], ha="center", va="center")

plt.show()

# ===================== CLASSIFICATION REPORT =====================
report = classification_report(y_true, y_pred, target_names=class_names, output_dict=True)
print(classification_report(y_true, y_pred, target_names=class_names))

# ===================== PRECISION / RECALL / F1 BAR =====================
metrics = ["precision", "recall", "f1-score"]
for m in metrics:
    plt.figure()
    plt.bar(class_names, [report[c][m] for c in class_names])
    plt.title(f"{m.upper()} per class")
    plt.ylim(0,1)
    plt.show()

# ===================== ROC CURVES =====================
y_true_bin = label_binarize(y_true, classes=range(NUM_CLASSES))
y_prob = np.array(y_prob)

plt.figure()
for i in range(NUM_CLASSES):
    fpr, tpr, _ = roc_curve(y_true_bin[:, i], y_prob[:, i])
    roc_auc = auc(fpr, tpr)
    plt.plot(fpr, tpr, label=f"{class_names[i]} (AUC={roc_auc:.2f})")

plt.plot([0,1],[0,1],'k--')
plt.xlabel("False Positive Rate")
plt.ylabel("True Positive Rate")
plt.title("ROC Curves")
plt.legend()
plt.grid()
plt.show()
